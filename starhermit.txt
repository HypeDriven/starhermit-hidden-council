# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com; never edit it). Re-run `npm run package` after every
# change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.22.0
engine.minor=19
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
control.pause=Escape | Pause / resume
control.undo=KeyU | Undo (practice)
control.hint=KeyH | Hint
control.camera=KeyC | Reset camera
control.wait=Space | Wait
control.commit=Enter+NumpadEnter | Go to focused room
control.prev_target=ArrowLeft+ArrowUp | Previous move target
control.next_target=ArrowRight+ArrowDown | Next move target
engine.group.pause=Session
engine.group.commit=Move targets

# Gamepad defaults for the same actions (the D-pad and South also move and
# press the UI's focus; the shoulders browse the move targets).
engine.gamepad.pause=Start
engine.gamepad.undo=West
engine.gamepad.hint=North
engine.gamepad.camera=RightStickPress
engine.gamepad.wait=Select
engine.gamepad.commit=RightTrigger
engine.gamepad.prev_target=LeftShoulder
engine.gamepad.next_target=RightShoulder
